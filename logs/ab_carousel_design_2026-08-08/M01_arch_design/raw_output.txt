// 信号日期落后超过 MAX_STALE_DAYS 的策略视为"未运行"，跳过渲染
const MAX_STALE_DAYS = 5;
function isStrategyActive(s) {
  const sid = s.id || s.strategy_id;
  const sigDate = s.metrics?.signal_date || (window.__todayActions && window.__todayActions[sid]?.signal_date) || '';
  if (!sigDate) return false; // 无信号日期 = 无数据 = 不在运行
  const now = new Date();
  const sig = new Date(sigDate + 'T00:00:00');
  const diffDays = Math.floor((now - sig) / (1000 * 60 * 60 * 24));
  return diffDays <= MAX_STALE_DAYS;
}
function activeStrategies(strategies) {
  return (strategies || []).filter(isStrategyActive);
}

// 动态生成策略卡片栅格
function renderStrategyCards(strategies) {
  const container = document.getElementById('strategy-cards');
  if (!container) return;
  const filtered = strategies ? activeStrategies(strategies) : [];
  if (filtered.length === 0) {
    container.innerHTML = '<div class="col-12" style="text-align:center;color:var(--text-muted);padding:48px">暂无正在运行的策略数据</div>';
    return;
  }

  const rows = calcGridRows(filtered.length);
  let html = '';
  let idx = 0;
  rows.forEach((rowSize, rowIdx) => {
    // 复用 row flex 布局
    html += '<div class="row strategy-row" data-row="' + (rowIdx + 1) + '">';
    for (let i = 0; i < rowSize && idx < filtered.length; i++, idx++) {
      const s = filtered[idx];
      const sid = s.id || s.strategy_id;
      const sname = s.name || s.strategy_name || sid;
      // 100% 自适应: 每张 col 宽度 = 100/rowSize
      const colClass = 'col-row-' + rowSize;
      const cardHtml = buildCardHtml(sid, sname, s);
      html += '<div class="' + colClass + '">' + cardHtml + '</div>';
    }
    html += '</div>';
  });
  container.innerHTML = html;
}
